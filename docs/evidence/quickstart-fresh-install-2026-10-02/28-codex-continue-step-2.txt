# Tiny command-line todo app

Use plain JavaScript for Node.js only; never use TypeScript. Store todos in `todos.json`, not a database, so the app runs with no setup.

1. Add a todo with `node todo.js add "Buy milk"`, creating `todos.json` automatically when needed.
2. List todos with `node todo.js list`.
3. Mark a todo done with `node todo.js done <id>`.
#!/usr/bin/env node
'use strict';

const fs = require('node:fs');
const path = require('node:path');

const dataFile = path.join(__dirname, 'todos.json');
const [command, ...words] = process.argv.slice(2);
const title = words.join(' ').trim();

if (command !== 'add' || !title) {
  console.error('Usage: node todo.js add "Todo description"');
  process.exitCode = 1;
} else {
  try {
    let todos = [];
    try {
      todos = JSON.parse(fs.readFileSync(dataFile, 'utf8'));
    } catch (error) {
      if (error.code !== 'ENOENT') throw error;
    }

    if (!Array.isArray(todos) || todos.some(todo =>
      !todo || !Number.isSafeInteger(todo.id) || todo.id < 1 ||
      typeof todo.title !== 'string' || typeof todo.done !== 'boolean'
    )) {
      throw new Error('todos.json must contain an array of valid todos.');
    }

    const id = todos.reduce((max, todo) => Math.max(max, todo.id), 0) + 1;
    if (!Number.isSafeInteger(id)) throw new Error('Todo ID limit reached.');
    todos.push({ id, title, done: false });
    fs.writeFileSync(dataFile, JSON.stringify(todos, null, 2) + '\n');
    console.log(`Added todo ${id}: ${title}`);
  } catch (error) {
    console.error(`Could not add todo: ${error.message}`);
    process.exitCode = 1;
  }
}
